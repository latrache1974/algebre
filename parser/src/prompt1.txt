Je veux écrire les régles de la grammaire entre {} exemple {$$->notation=$1 ...}
La grammaire est la suivante:

S : IncludeSection Algebre
  ;

IncludeSection : INCLUDE STRING IncludeSection {Utils::PerformeInclude($2);}
               | /* vide */
               ;

Algebre : TypeDefinition Algebre {universe->AddType($1);}
        | FunctionDefinition Algebre {universe->AddFunction($1);}
        | LiteralDefinition Algebre  {universe->AddLiteral($1);}
        | TheoremeDefinition Algebre {universe->AddTheoreme($1);}
        | GoalDefinition END '.' {universe->SetGoal($1);}
        ;

TypeDefinition : TYPE IDENT '(' DoubleIdentList ')' ';' NotationDefinition VariablesDefinitions Corps
                 {$$=new Type();
                 $$->name=$2;
                 $$->AssignFromDoubleIdentList($4);
                 delete $4;
                 $$->notation=$7;
                 $$->Vars=$8;
                 $$->AssignFromCorps($9);}
               ;

LiteralDefinition : LITERAL ':' IDENT ';' NotationDefinition VariablesDefinitions Corps
                 {$$=new Literal();
                 $$->type=$3;
                 $$->notation=$5;
                 $$->Vars=$6;
                 $$->AssignFromCorps($7);}
                  ;

FunctionDefinition : FUNCTION IDENT '(' DoubleIdentList ')' ';' NotationDefinition VariablesDefinitions Corps
                 {$$=new Function();
                 $$->name=$2;
                 $$->AssignFromDoubleIdentList($4);
                 $$->notation=$7;
                 $$->Vars=$8;
                 $$->AssignFromCorps($9);}
                   ;

TheoremeDefinition : THEOREME STRING '(' DoubleIdentList ')' ';' VariablesDefinitions CorpsM
                 {$$=new Theoreme();
                 $$->name=$2;
                 $$->AssignFromDoubleIdentList($4);
                 $$->Vars=$7;
                 $$->AssignFromCorpsM($8);}
                   ;

GoalDefinition : VariablesDefinitions CorpsM
                    {$$=new Goal();
                    $$->Vars=$1;
                    $$->AssignFromCorpsM($2);}
               | { }
               ;

NotationDefinition : NOTATION STRING NameSpaceList ';'
                    {$$=new Notation();
                    $$->format=$2;
                    $$->params=new NameSpaceList();}
                   | {$$=NULL;}
                   ;

NameSpaceList : NameSpaceList1 {$$=$1;}
              | {}
              ;
NameSpaceList1 : NAMESPACE ',' NameSpaceList1 {$$->Add($1);}
              | NAMESPACE  {$$->Add($1);}
              ;

DoubleIdentList : DoubleIdentList1 {$$=new Parameters(); }
                | /* vide */ {$$=NULL;}
                ;

DoubleIdentList1 : IDENT IDENT ',' DoubleIdentList1 {$$->Add($1,$2);}
                 | IDENT IDENT {$$->Add($1,$2);}
                 ;

VariablesDefinitions : VAR VariablesDefinitions1 {$$=new Variables();}
                     | /* vide */ {$$=NULL;}
                     ;

VariablesDefinitions1 : QUANTIFIER IDENT IDENT ';' VariablesDefinitions1 {$$->Add($1, $2, $3);}
                      | QUANTIFIER IDENT IDENT ';' {$$->Add($1, $2, $3);}
                      ;

/* Version mise à jour de Corps */
Corps : Predicat1 ARROW_IMPLIES Predicat ';' { $$=new Corps(); $$->Assign($1, "=>", $3);}
      | Predicat1 ARROW_EQUIV Predicat1 ';' { $$=new Corps(); $$->Assign($1, "<=>", $3);}
      | Predicat1 ';'  { $$=new Corps(); $$->Assign($1);}
      ;

/* Version specifique aux Théorèmes et Objectifs */
CorpsM : Predicat1 ARROW_IMPLIES Predicats ';' {
            $$=new CorpsM();
            $$->left=$1;
            $$->sentences=$3;
            $$->eq=false;}
       | Predicat1 ARROW_EQUIV Predicat1 ';' {
            $$=new CorpsM();
            $$->left=$1;
            $$->right=$3;
            $$->eq=true;}

       | Predicats ';'  {
            $$=new CorpsM();
            $$->sentences=$1; }
       ;

/* Liste de prédicats séparés par des barres verticales '|' */
Predicats : Predicats2 {$$=$1;}
          | {}
          ;
Predicats2 : Predicat '|' Predicats {$$->Add($1); }
          | Predicat {$$->Add($1); }
          ;

Predicat1 : Atom {$$=$1;}
          | Predicat {$$=$1;}
          ;

Predicat : NOT Predicat  {$$=Utils::CreateNot($2);}
         | Predicat OR Predicat {$$=Utils::CreateOr($1,$3);}
         | Predicat AND Predicat {$$=Utils::CreateAnd($1,$3);}
         | Predicat XOR Predicat {$$=Utils::CreateXor($1,$3);}
         | '(' Atom ')' {$$=$2;}
         ;

Atom : IDENT '(' Atoms ')'  {$$=Utils::CreateFunction($1,$3);}
     | IDENT {$$=Utils::CreateTerminal($1);}
     | APPARTIENT '(' IDENT ',' IDENT ')' {$$=Utils::CreateAppartient($3,$5);}
     | RELATION '(' IDENT ',' IDENT ')' {$$=Utils::CreateRelation($3,$5);}
     ;

Atoms : Atom ',' Atoms {$$->Add($1);}
      | Atom {$$->Add($1);}
      ;
