
function application(set A, set B);
notation "%0", instance.name;
var _A x1 element, appartient(x1, A);
    _A x2 element, appartient(x2, A);
    _A y element, appartient(y, B);
begin
  relation(x1,y) and relation(x2,y) => x1=x2;
end;

type lci(application l, set A);
notation "%0", instance.name;
var _A x element, appartient(x, A);
    _A y element, appartient(y, A);
begin
  => appartient(l(x,y), A);
end;

literal  : set;
notation "∅"
var _A x element;
begin
  => not appartient(x, ∅);
end

theoreme "Récursivité de l'inclusion" (set A, set B, set C);
begin
  inclus(A, B) and inclus(B, C) => inclus(A, C);
end
